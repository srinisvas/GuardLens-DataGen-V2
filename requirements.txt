# GuardLens DataGen V2 — pip requirements
#
# Target system:
#   NVIDIA A100 80GB PCIe (Ampere, sm_80)
#   Driver R580 LTS (580.105.08+), CUDA runtime 13.0
#   Linux x86_64
#
# Recommended env creation:
#   conda create -n dataset_gen python=3.11 -y
#   conda activate dataset_gen
#   pip install -r requirements.txt
#
# CUDA compatibility notes:
#   Driver 580 supports CUDA 13.0 natively. PyTorch >=2.11 and vLLM >=0.20
#   both publish CUDA 13.0 wheels as the PyPI default, so no --extra-index-url
#   is required. If you have a mixed cluster where some nodes are on older
#   drivers, you can pin to CUDA 12.9 wheels instead:
#     pip install torch --index-url https://download.pytorch.org/whl/cu129
#   then install this file. vLLM will auto-detect the installed torch's CUDA.
#
# Install order matters:
#   vLLM pins its own torch version. Install this file top-to-bottom OR let
#   pip resolve the full graph in one shot. Do NOT install torch, then vllm,
#   then a different torch — pip will silently downgrade.
